        function setupFirestoreSync() {

            clearFirestoreSync();


            const paths = [

                { key: 'incomeStreams', name: 'incomeStreams' },

                { key: 'fixedBills', name: 'fixedBills' },

                { key: 'categoryBudgets', name: 'categoryBudgets' },

                { key: 'transactions', name: 'transactions' },

                { key: 'projects', name: 'projects' },

                { key: 'projectExpenses', name: 'projectExpenses' },

                { key: 'savingsGoals', name: 'savingsGoals' }

            ];


            paths.forEach(p => {

                const colRef = collection(db, 'artifacts', appId, 'public', 'data', p.name);

                const unsub = onSnapshot(colRef, (snapshot) => {

                    let items = [];

                    snapshot.forEach(docSnap => {

                        items.push({ id: docSnap.id, ...docSnap.data() });

                    });

                    

                    if (p.key === 'categoryBudgets') {

                        if (items.length > 0) {

                            state.categoryBudgets = items[0].budgets || state.categoryBudgets;

                        }

                    } else {

                        state[p.key] = items;

                    }

                    renderAll();

                }, (err) => {

                    console.error("Firestore sync error on " + p.name, err);

                });

                unsubscribeListeners.push(unsub);

            });

        }


        function clearFirestoreSync() {

            unsubscribeListeners.forEach(unsub => unsub());

            unsubscribeListeners = [];

        }

        function getIncomeForMonth(monthStr) {

            const [year, month] = monthStr.split('-').map(Number);

            const targetDate = new Date(year, month, 0);


            let total = 0;

            state.incomeStreams.forEach(stream => {

                if (!stream.startDate) return;

                const startDate = new Date(stream.startDate);

                if (startDate <= targetDate) {

                    total += Number(stream.amount || 0);

                }

            });

            return total;

        }


        function getFixedBillsTotal() {

            return state.fixedBills.reduce((sum, bill) => sum + Number(bill.amount || 0), 0);

        }


        function getVariableSpentForMonth(monthStr) {

            return state.transactions

                .filter(tx => tx.date && tx.date.startsWith(monthStr))

                .reduce((sum, tx) => sum + Number(tx.amount || 0), 0);

        }


        // Calculate accumulated surplus / rollover across all historical recorded months

        function getAccumulatedRollover() {

            // Find all unique months in transactions or income streams

            let monthsSet = new Set();

            state.transactions.forEach(tx => {

                if (tx.date) monthsSet.add(tx.date.slice(0, 7));

            });

            monthsSet.add(state.currentMonth);


            let totalAccumulated = 0;

            monthsSet.forEach(m => {

                const inc = getIncomeForMonth(m);

                const fix = getFixedBillsTotal();

                const varSp = getVariableSpentForMonth(m);

                totalAccumulated += (inc - fix - varSp);

            });

            return totalAccumulated;

        }

        function renderCategoryProgress(month) {

            const container = document.getElementById('category-progress-list');

            const categories = Object.keys(state.categoryBudgets);

            

            const spentMap = {};

            categories.forEach(cat => spentMap[cat] = 0);

            

            state.transactions

                .filter(tx => tx.date && tx.date.startsWith(month))

                .forEach(tx => {

                    if (spentMap[tx.category] !== undefined) {

                        spentMap[tx.category] += Number(tx.amount || 0);

                    } else {

                        spentMap[tx.category] = Number(tx.amount || 0);

                    }

                });


            if (categories.length === 0) {

                container.innerHTML = `<p class="text-sm text-slate-400 text-center py-4">No categories configured.</p>`;

                return;

            }

                const spent = spentMap[cat] || 0;

                const pct = limit > 0 ? Math.min(Math.round((spent / limit) * 100), 100) : 0;

                const isOver = spent > limit && limit > 0;

                

                let barColor = 'bg-brand-500';

                if (pct >= 80 && pct < 100) barColor = 'bg-amber-500';

                if (pct >= 100 || isOver) barColor = 'bg-red-500';


                return `

                    <div class="space-y-1.5">

                        <div class="flex justify-between items-center text-sm">

                            </span>

                                <strong class="${isOver ? 'text-red-500 font-bold' : ''}">${formatCurrency(spent)}</strong> / ${formatCurrency(limit)} (${pct}%)

        window.convertGoalToProject = async function(goalId) {

            const goal = state.savingsGoals.find(g => g.id === goalId);

            if (!goal || !currentUser) return;


            if (!confirm(`Convert "${goal.name}" into an active Project Cost Tracker? This will create a project with a budget of ${formatCurrency(goal.targetAmount)}.`)) return;


            try {

                // 1. Create project

                const projCol = collection(db, 'artifacts', appId, 'public', 'data', 'projects');

                const newProjRef = await addDoc(projCol, {

                    name: goal.name + " (Converted)",

                    totalBudget: goal.targetAmount,

                    categories: goal.categories || ["General"],

                    createdAt: new Date().toISOString()

                });


                // 2. If already saved money, log initial starting expense/draw or seed record

                if (goal.currentSaved > 0) {

                    const peCol = collection(db, 'artifacts', appId, 'public', 'data', 'projectExpenses');

                    await addDoc(peCol, {

                        projectId: newProjRef.id,

                        date: new Date().toISOString().slice(0, 10),

                        category: goal.categories[0] || "General",

                        amount: goal.currentSaved,

                        vendor: "Initial Savings Transfer",

                        source: "Savings Goal Conversion",

                        createdAt: new Date().toISOString()

                    });

                }


                // 3. Delete savings goal

                const goalDocRef = doc(db, 'artifacts', appId, 'public', 'data', 'savingsGoals', goalId);

                await deleteDoc(goalDocRef);


                showToast("Successfully converted goal to project tracker!");

                switchMainTab('projects');

            } catch (err) {

                showToast("Error converting goal: " + err.message, "error");

            }

        };

        // --- Utility Helpers ---


        function formatCurrency(amount) {

            return new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD' }).format(amount || 0);

        }


        function escapeHtml(str) {

            if (!str) return '';

            return str.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;").replace(/'/g, "&#039;");

        }


        function showToast(message, type = 'success') {

            const toast = document.getElementById('toast');

            const toastMsg = document.getElementById('toast-message');

            const toastIcon = document.getElementById('toast-icon');


            toastMsg.innerText = message;

            if (type === 'error') {

                toastIcon.className = "fa-solid fa-circle-exclamation text-red-400";

            } else {

                toastIcon.className = "fa-solid fa-circle-check text-brand-400";

            }


            toast.classList.remove('translate-y-20', 'opacity-0');

            setTimeout(() => {

                toast.classList.add('translate-y-20', 'opacity-0');

            }, 3500);

        }
